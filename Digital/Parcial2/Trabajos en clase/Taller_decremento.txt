@ Equivalente ARM para cpulator (?sys=arm)
@ Cuenta regresiva de 99 a 0 con paso seleccionable (1 o 2)
@ R4 = M (decena), R5 = N (unidad), R6 = paso
.global _start
_start:
	@ Preguntar 1 o 2
	ADR R0, pregunta
	BL put_string        @ lea dx, pregunta / int 21h
	BL put_nl            @ salto de linea
	@ Leer opcion
	BL get_char          @ mov ah, 01h / int 21h
	MOV R6, #2           @ default: paso = 2
	CMP R0, #'1'         @ cmp al, '1'
	BEQ dec_uno          @ je dec_uno
	B inicio             @ jmp inicio
dec_uno:
	MOV R6, #1           @ mov paso, 1

inicio:
	MOV R4, #9           @ mov M, 9
	BL put_nl
ciclo2:
	MOV R5, #9           @ mov N, 9
ciclo1:
	@ imprimir M (decena)
	MOV R1, R4           @ mov dl, M
	ADD R1, R1, #48      @ add dl, 48
	BL put_char          @ mov ah, 2 / int 21h
	@ imprimir N (unidad)
	MOV R1, R5           @ mov dl, N
	ADD R1, R1, #48      @ add dl, 48
	BL put_char          @ mov ah, 2 / int 21h
	@ salto de linea
	BL put_nl
	@ decrementar N segun paso
	MOV R0, R5           @ mov al, N
	SUB R0, R0, R6       @ sub al, paso
	MOV R5, R0           @ mov N, al
	CMP R0, #0           @ jns ciclo1 (saltar si >= 0)
	BPL ciclo1
	@ decrementar M segun paso (siempre en 1)
	MOV R0, R4           @ mov al, M
	SUB R0, R0, #1       @ sub al, 1
	MOV R4, R0           @ mov M, al
	CMP R0, #0           @ jns ciclo2
	BPL ciclo2
	CMP R0, #0           @ jnz inicio
	BNE inicio
_stop:
	B _stop              @ ret

@ Subrutinas de E/S via semihosting
put_nl:
	STMFD SP!, {LR}      @ guardar direccion de retorno
	MOV R1, #10          @ LF
	BL put_char
	MOV R1, #13          @ CR
	BL put_char
	LDMFD SP!, {LR}      @ restaurar direccion de retorno
	BX LR

put_char:
	SUB SP, SP, #4       @ espacio para el caracter
	STRB R1, [SP]        @ guardar caracter en pila
	MOV R0, #0x03        @ SYS_WRITEC
	MOV R1, SP           @ R1 = puntero al caracter
	SVC #0x123456
	ADD SP, SP, #4       @ restaurar pila
	BX LR

put_string:
	MOV R1, R0           @ R1 = puntero al string
	MOV R0, #0x04        @ SYS_WRITE0
	SVC #0x123456
	BX LR

get_char:
	MOV R0, #0x07        @ SYS_READC
	SVC #0x123456
	BX LR

pregunta: .asciz "Cuanto quieres que decremente en 1 o en 2? "
